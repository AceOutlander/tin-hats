phase('Missions')
const reports = []
for (let i = 0; i < MISSIONS.length; i += 2) {
  const pair = MISSIONS.slice(i, i + 2)
  const r = await parallel(pair.map(id => () => agent(missionTask(id), { label: `mission:${id}`, phase: 'Missions' }).then(rep => ({ id, rep }))))
  reports.push(...r.filter(Boolean))
  log(`Missions done: ${reports.map(r => r.id).join(', ')}`)
}
